name: Build Android APK

on:
  workflow_dispatch:
  push:
    branches: [ main ]

env:
  APP_URL: "https://PASTE-YOUR-APP-ADDRESS-HERE"

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 22

      - uses: actions/setup-java@v4
        with:
          distribution: zulu
          java-version: 21

      - name: Check address
        run: |
          if echo "$APP_URL" | grep -q PASTE; then
            echo "APP_URL is not set. Edit build.yml first."
            exit 1
          fi

      - name: Create Android shell
        run: |
          mkdir -p shell/www
          cp native-shell/index.html shell/www/index.html
          cd shell
          npm init -y
          npm install @capacitor/core @capacitor/cli @capacitor/android
          printf '{"appId":"com.mira.sweetheart","appName":"Mira","webDir":"www","server":{"url":"%s","cleartext":false}}' "$APP_URL" > capacitor.config.json
          npx cap add android

      - name: Android permissions
        run: |
          cd shell
          MANIFEST=android/app/src/main/AndroidManifest.xml
          sed -i 's#<application#<uses-permission android:name="android.permission.RECORD_AUDIO" />\n    <uses-permission android:name="android.permission.MODIFY_AUDIO_SETTINGS" />\n    <application#' $MANIFEST
          grep -n "uses-permission" $MANIFEST
          npx cap sync android

      - name: Build APK
        run: |
          cd shell/android
          chmod +x gradlew
          ./gradlew assembleDebug

      - uses: actions/upload-artifact@v4
        with:
          name: Mira-App
          path: shell/android/app/build/outputs/apk/debug/app-debug.apk
